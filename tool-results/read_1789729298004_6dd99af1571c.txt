     9→/**
    10→ * GET /api/report
    11→ * Query params:
    12→ *   mode          - "all" | "order" | "bku"   (default: all)
    13→ *   orderId       - single order id (for mode=order)
    14→ *   bku           - BKU code substring (for mode=bku)
    15→ *   noPesanan     - No Pesanan substring (alternative to orderId)
    16→ *   includePhotos - "1" | "0"  (default: 1)
    17→ *   includeItems  - "1" | "0"  (default: 1)
    18→ *   format        - "full" | "lampiran"  (default: full)
    19→ *                   - "full"     = laporan lengkap (data + tabel + foto)
    20→ *                   - "lampiran" = Lampiran Gambar BAST (photo-only, match reference PDF)
    21→ *
    22→ * Returns a complete HTML document (print-ready, A4) that auto-triggers print().
    23→ */
    24→export async function GET(req: NextRequest) {
    25→  try {
    26→    const { searchParams } = new URL(req.url);
    27→    const mode = (searchParams.get("mode") || "all") as
    28→      | "all"
    29→      | "order"
    30→      | "bku";
    31→    const orderId = searchParams.get("orderId") || "";
    32→    const bku = (searchParams.get("bku") || "").trim();
    33→    const noPesanan = (searchParams.get("noPesanan") || "").trim();
    34→    const includePhotos = searchParams.get("includePhotos") !== "0";
    35→    const includeItems = searchParams.get("includeItems") !== "0";
    36→    const format = (searchParams.get("format") || "full") as
    37→      | "full"
    38→      | "lampiran";
    39→    // Date range filter (for mode=all)
    40→    const startDate = searchParams.get("startDate") || "";
    41→    const endDate = searchParams.get("endDate") || "";
    42→
    43→    // Build where clause
    44→    const where: Record<string, unknown> = {};
    45→    if (mode === "order" && orderId) {
    46→      where.id = orderId;
    47→    } else if (mode === "order" && noPesanan) {
    48→      where.noPesanan = noPesanan;
    49→    } else if (mode === "bku" && bku) {
    50→      where.noBku = { contains: bku };
    51→    }
    52→
    53→    // Date range filter: filter by tanggalPesanan (format DD/MM/YYYY in Excel)
    54→    // We filter in-memory after fetch since tanggalPesanan is stored as String (not Date)
    55→    // This is handled after fetch below
    56→
    57→    // Fetch orders with items, legacy photos (SpjPhoto), and documentation photos (via PhotoOrderLink)
    58→    const orders = await db.spjOrder.findMany({
    59→      where,
    60→      include: {
    61→        items: { orderBy: { createdAt: "asc" } },
    62→        photos: { orderBy: { createdAt: "asc" } },
    63→        photoLinks: {
    64→          include: { photo: true },
    65→          orderBy: { createdAt: "asc" },
    66→        },
    67→      },
    68→      orderBy: [{ noPesanan: "asc" }],
    69→    });
    70→
    71→    // ===== Date range filter (in-memory, since tanggalPesanan is String "DD/MM/YYYY") =====
    72→    // Parse "DD/MM/YYYY" to Date object for comparison
    73→    function parseDateStr(s: string | null): Date | null {
    74→      if (!s) return null;
    75→      const parts = s.split("/");
    76→      if (parts.length < 3) return null;
    77→      const d = parseInt(parts[0], 10);
    78→      const m = parseInt(parts[1], 10) - 1;
    79→      let y = parseInt(parts[2], 10);
    80→      if (y < 100) y += 2000;
    81→      return new Date(y, m, d);
    82→    }
    83→
    84→    let filteredOrders = orders;
    85→    if (startDate || endDate) {
    86→      const start = startDate ? new Date(startDate + "T00:00:00") : null;
    87→      const end = endDate ? new Date(endDate + "T23:59:59") : null;
    88→
    89→      filteredOrders = orders.filter((o) => {
    90→        const orderDate = parseDateStr(o.tanggalPesanan);
    91→        if (!orderDate) return false; // skip if date can't be parsed
    92→        if (start && orderDate < start) return false;
    93→        if (end && orderDate > end) return false;
    94→        return true;
    95→      });
    96→    }
    97→
    98→    if (filteredOrders.length === 0) {
    99→      return NextResponse.json(
   100→        { error: "Tidak ada data yang cocok dengan filter" },
   101→        { status: 404 }
   102→      );
   103→    }
   104→
   105→    // Numeric sort
   106→    filteredOrders.sort((a, b) => {
   107→      const na = parseInt(a.noPesanan, 10);
   108→      const nb = parseInt(b.noPesanan, 10);
   109→      if (!Number.isNaN(na) && !Number.isNaN(nb) && na !== nb) return na - nb;
   110→      return a.noPesanan.localeCompare(b.noPesanan);
   111→    });
   112→
   113→    // Compute summary
   114→    const totalItems = filteredOrders.reduce(
   115→      (acc, o) => acc + o.items.length,
   116→      0
   117→    );
   118→    const totalPhotos = filteredOrders.reduce(
   119→      (acc, o) => acc + o.photos.length,
   120→      0
   121→    );
   122→    const grandTotal = filteredOrders.reduce((acc, o) => {
   123→      return (
   124→        acc +
   125→        o.items.reduce((s, it) => {
   126→          const n = Number(String(it.jumlah ?? "0").replace(/[^\d.-]/g, ""));
   127→          return s + (Number.isNaN(n) ? 0 : n);
   128→        }, 0)
   129→      );
   130→    }, 0);
   131→
   132→    const generatedAt = new Date().toLocaleString("id-ID", {
   133→      dateStyle: "long",
   134→      timeStyle: "short",
   135→    });
   136→
   137→    // ===== LAMPIRAN FORMAT (match reference PDF: Gambar BAST PC all in one.pdf) =====
   138→    if (format === "lampiran") {
   139→      // Construct base URL for footer (absolute URL like reference PDF)
   140→      const host = req.headers.get("host") || "localhost:3000";
   141→      const protocol = req.headers.get("x-forwarded-proto") || "http";
   142→      const baseUrl = `${protocol}://${host}`;
   143→
   144→      // Read actual image dimensions for orientation detection
   145→      // Merge photos from TWO sources:
   146→      //   1. Legacy SpjPhoto (o.photos) — old upload system
   147→      //   2. DocumentationPhoto via PhotoOrderLink (o.photoLinks) — new many-to-many system
   148→      // Deduplicate by filePath to avoid showing the same photo twice
   149→      // (migration may have created both SpjPhoto and DocumentationPhoto for same file)
   150→      const ordersWithMeta = await Promise.all(
   151→        filteredOrders.map(async (o) => {
   152→          // Collect all unique photo URLs from both sources
   153→          const seenUrls = new Set<string>();
   154→          const allPhotos: Array<{
   155→            id: string;
   156→            url: string;
   157→            fileName: string;
   158→            filePath: string;
   159→            deviceType: string;
   160→            source: string;
   161→            orientation: ImageOrientation;
   162→          }> = [];
   163→
   164→          // Source 1: Legacy SpjPhoto
   165→          for (const p of o.photos) {
   166→            if (seenUrls.has(p.url)) continue;
   167→            seenUrls.add(p.url);
   168→            const orientation = await getImageOrientation(p.filePath);
   169→            allPhotos.push({
   170→              id: p.id,
   171→              url: p.url,
   172→              fileName: p.fileName,
   173→              filePath: p.filePath,
   174→              deviceType: p.deviceType,
   175→              source: p.source,
   176→              orientation,
   177→            });
   178→          }
   179→
   180→          // Source 2: DocumentationPhoto via PhotoOrderLink (many-to-many)
   181→          for (const link of o.photoLinks) {
   182→            const p = link.photo;
   183→            if (seenUrls.has(p.url)) continue;
   184→            seenUrls.add(p.url);
   185→            const orientation = await getImageOrientation(p.filePath);
   186→            allPhotos.push({
   187→              id: p.id,
   188→              url: p.url,
   189→              fileName: p.fileName,
   190→              filePath: p.filePath,
   191→              deviceType: p.deviceType,
   192→              source: p.source,
   193→              orientation,
   194→            });
   195→          }
   196→
   197→          return {
   198→            ...o,
   199→            photos: allPhotos, // merged + deduplicated photos
   200→          };
   201→        })
   202→      );
   203→
   204→      const lampiranHtml = renderLampiran({
   205→        mode,
   206→        bku,
   207→        orders: ordersWithMeta,
   208→        summary: {
   209→          totalOrders: filteredOrders.length,
   210→          totalItems,
   211→          totalPhotos,
   212→          grandTotal,
   213→        },
   214→        generatedAt,
   215→        baseUrl,
   216→      });
   217→
   218→      return new NextResponse(lampiranHtml, {
   219→        headers: {
   220→          "Content-Type": "text/html; charset=utf-8",
   221→          "Cache-Control": "no-store, no-cache, must-revalidate",
   222→        },
   223→      });
   224→    }
   225→
   226→    // ===== FULL FORMAT (default, existing report) =====
   227→    const title =
   228→      mode === "order"
   229→        ? `Laporan SPJ — No. Pesanan #${orders[0].noPesanan}`
   230→        : mode === "bku"
   231→        ? `Laporan SPJ — BKU: ${bku}`
   232→        : "Laporan SPJ — Semua No. Pesanan";
   233→
   234→    const reportHtml = renderReport({
   235→      title,
   236→      mode,
   237→      orders,
   238→      includePhotos,
   239→      includeItems,
   240→      summary: {
   241→        totalOrders: filteredOrders.length,
   242→        totalItems,
   243→        totalPhotos,
   244→        grandTotal,
   245→      },
   246→      generatedAt,
   247→    });
   248→
   249→    return new NextResponse(reportHtml, {
   250→      headers: {
   251→        "Content-Type": "text/html; charset=utf-8",
   252→        "Cache-Control": "no-store, no-cache, must-revalidate",
   253→      },
   254→    });
   255→  } catch (err) {
   256→    console.error("GET /api/report error:", err);
   257→    return NextResponse.json(
   258→      { error: "Gagal membuat laporan" },
   259→      { status: 500 }
   260→    );
   261→  }
   262→}

   310→function formatRupiah(val: string | null | undefined): string {
   311→  if (!val) return "-";
   312→  const n = Number(String(val).replace(/[^\d.-]/g, ""));
   313→  if (Number.isNaN(n)) return val;
   314→  return new Intl.NumberFormat("id-ID", {
   315→    style: "currency",
   316→    currency: "IDR",
   317→    maximumFractionDigits: 0,
   318→  }).format(n);
   319→}
   320→
   321→function esc(s: string | null | undefined): string {
   322→  if (!s) return "";
   323→  return String(s)
   324→    .replace(/&/g, "&amp;")
   325→    .replace(/</g, "&lt;")
   326→    .replace(/>/g, "&gt;")
   327→    .replace(/"/g, "&quot;")
   328→    .replace(/'/g, "&#39;");
   329→}

   901→type ImageOrientation = "landscape" | "portrait" | "square" | "unknown";
   902→
   903→/**
   904→ * Read actual image dimensions from file on disk using sharp.
   905→ * Returns orientation:
   906→ *   - landscape  (w / h > 1.15)  → card 4:3 landscape
   907→ *   - portrait   (h / w > 1.15)  → card 3:4 portrait
   908→ *   - square     (0.87 ≤ ratio ≤ 1.15) → card 1:1
   909→ *   - unknown    (cannot read)   → fallback to landscape
   910→ *
   911→ * By matching the card aspect ratio to the photo's natural orientation,
   912→ * object-fit: cover produces minimal crop — important parts stay visible.
   913→ */
   914→async function getImageOrientation(
   915→  filePath: string
   916→): Promise<ImageOrientation> {
   917→  try {
   918→    const absPath = join(process.cwd(), "public", filePath);
   919→    const meta = await sharp(absPath).metadata();
   920→    const w = meta.width ?? 0;
   921→    const h = meta.height ?? 0;
   922→    if (w === 0 || h === 0) return "unknown";
   923→    const ratio = w / h;
   924→    if (ratio >= 0.87 && ratio <= 1.15) return "square";
   925→    if (ratio > 1.15) return "landscape";
   926→    return "portrait";
   927→  } catch {
   928→    return "unknown";
   929→  }
   930→}

  1222→/**
  1223→ * Convert month number (1-12) to Roman numeral.
  1224→ * 1=I, 2=II, 3=III, 4=IV, 5=V, 6=VI, 7=VII, 8=VIII, 9=IX, 10=X, 11=XI, 12=XII
  1225→ */
  1226→function monthToRoman(month: number): string {
  1227→  const romans = [
  1228→    "I", "II", "III", "IV", "V", "VI",
  1229→    "VII", "VIII", "IX", "X", "XI", "XII",
  1230→  ];
  1231→  if (month >= 1 && month <= 12) return romans[month - 1];
  1232→  return "I"; // fallback
  1233→}

  1235→/**
  1236→ * Parse tanggalBast from Excel format "DD/MM/YYYY" and extract Roman month + year.
  1237→ * Returns { romanMonth, year } for BAST number format.
  1238→ * Fallback to current date if parse fails.
  1239→ */
  1240→function parseTanggalBast(tanggalBast: string | null): {
  1241→  romanMonth: string;
  1242→  year: string;
  1243→} {
  1244→  if (!tanggalBast) {
  1245→    const now = new Date();
  1246→    return {
  1247→      romanMonth: monthToRoman(now.getMonth() + 1),
  1248→      year: String(now.getFullYear()),
  1249→    };
  1250→  }
  1251→  // Format: "DD/MM/YYYY" or "D/M/YY" or "D/M/YYYY"
  1252→  const parts = tanggalBast.split("/");
  1253→  if (parts.length < 3) {
  1254→    const now = new Date();
  1255→    return {
  1256→      romanMonth: monthToRoman(now.getMonth() + 1),
  1257→      year: String(now.getFullYear()),
  1258→    };
  1259→  }
  1260→  const month = parseInt(parts[1], 10);
  1261→  let year = parts[2].trim();
  1262→  // Handle 2-digit year (e.g. "25" → "2025")
  1263→  if (year.length === 2) year = "20" + year;
  1264→  // Validate
  1265→  if (Number.isNaN(month) || month < 1 || month > 12) {
  1266→    const now = new Date();
  1267→    return {
  1268→      romanMonth: monthToRoman(now.getMonth() + 1),
  1269→      year: String(now.getFullYear()),
  1270→    };
  1271→  }
  1272→  return { romanMonth: monthToRoman(month), year };
  1273→}

  1275→function renderLampiranOrderSection(
  1276→  o: LampiranOrder,
  1277→  pageNum: number,
  1278→  totalPages: number,
  1279→  baseUrl: string
  1280→): string {
  1281→  // Format BAST number: NOMOR : 421.3/{noPesanan}-BAST/SMANSA-TD/{bulan-romawi}/{tahun}
  1282→  // - 421.3/ = prefix tetap
  1283→  // - {noPesanan} = dari Excel (contoh: 45)
  1284→  // - -BAST/SMANSA-TD/ = middle tetap
  1285→  // - {bulan-romawi} = otomatis dari tanggalBast Excel (I-XII)
  1286→  // - {tahun} = otomatis dari tahun tanggalBast Excel
  1287→  const { romanMonth, year } = parseTanggalBast(o.tanggalBast);
  1288→  const bastNumber = `421.3/${o.noPesanan}-BAST/SMANSA-TD/${romanMonth}/${year}`;
  1289→
  1290→  // Source URL (absolute, matches reference PDF footer format)
  1291→  const sourceUrl = `${baseUrl}/api/report?mode=order&noPesanan=${encodeURIComponent(
  1292→    o.noPesanan
  1293→  )}&format=lampiran`;
  1294→
  1295→  // Initial timestamp (will be updated by JS for live time)
  1296→  const now = new Date();
  1297→  const ts = `${String(now.getDate()).padStart(2, "0")}/${String(
  1298→    now.getMonth() + 1
  1299→  ).padStart(2, "0")}/${String(now.getFullYear()).slice(-2)}, ${String(
  1300→    now.getHours()
  1301→  ).padStart(2, "0")}.${String(now.getMinutes()).padStart(2, "0")}`;
  1302→
  1303→  // ===== Smart photo ordering for professional layout =====
  1304→  // Group by orientation: "wide" (landscape/square/unknown) vs "tall" (portrait)
  1305→  // Strategy: MAJORITY group goes first (top rows, 2 columns),
  1306→  //           MINORITY group goes last (bottom row, centered if odd count).
  1307→  //
  1308→  // Examples:
  1309→  //   3 landscape + 1 portrait  → [L,L,L, P]      (landscape 2+1, portrait 1 centered)
  1310→  //   1 landscape + 3 portrait  → [P,P,P, L]      (portrait 2+1, landscape 1 centered)
  1311→  //   2 landscape + 2 portrait  → [L,L, P,P]      (landscape 2, portrait 2)
  1312→  //   2 landscape + 1 portrait  → [L,L, P]        (landscape 2, portrait 1 centered)
  1313→  //   1 landscape + 2 portrait  → [P,P, L]        (portrait 2, landscape 1 centered)
  1314→  //
  1315→  // All cards use the SAME 4:3 aspect ratio (uniform), so no card towers above others.
  1316→  const widePhotos = o.photos.filter(
  1317→    (p) => (p.orientation || "unknown") !== "portrait"
  1318→  );
  1319→  const tallPhotos = o.photos.filter(
  1320→    (p) => p.orientation === "portrait"
  1321→  );
  1322→
  1323→  // Majority first → ensures top rows are filled (2 cols), minority last
  1324→  // If equal count, default to wide (landscape) first (more natural for BAST)
  1325→  const sortedPhotos =
  1326→    tallPhotos.length > widePhotos.length
  1327→      ? [...tallPhotos, ...widePhotos]
  1328→      : [...widePhotos, ...tallPhotos];
  1329→
  1330→  const photosHtml =
  1331→    sortedPhotos.length === 0
  1332→      ? '<div class="no-photos">Belum ada foto dokumentasi untuk No. Pesanan ini.</div>'
  1333→      : `<div class="photo-grid">
  1334→          ${sortedPhotos
  1335→            .map((p, i) => {
  1336→              // If total photos is odd AND this is the last photo, center it
  1337→              const isOddLast =
  1338→                sortedPhotos.length % 2 === 1 &&
  1339→                i === sortedPhotos.length - 1;
  1340→              const classes = ["photo-card"];
  1341→              if (isOddLast) classes.push("last-odd");
  1342→              return `<div class="${classes.join(" ")}">
  1343→              <img src="${esc(p.url)}" alt="Foto ${i + 1}" />
  1344→            </div>`;
  1345→            })
  1346→            .join("")}
  1347→        </div>`;
  1348→
  1349→  return `<div class="order-section">
  1350→    <div class="top-header">
  1351→      <span class="timestamp">${esc(ts)}</span>
  1352→      <span class="doc-label">Gambar BAST</span>
  1353→    </div>
  1354→    <div class="divider"></div>
  1355→
  1356→    <h1 class="doc-title">Lampiran Gambar BAST</h1>
  1357→    <div class="doc-ref">NOMOR : ${esc(bastNumber)}</div>
  1358→
  1359→    ${photosHtml}
  1360→
  1361→    <div class="page-footer">
  1362→      <span class="url">${esc(sourceUrl)}</span>
  1363→      <span class="page-num">${pageNum}/${totalPages}</span>
  1364→    </div>
  1365→  </div>`;
  1366→}
